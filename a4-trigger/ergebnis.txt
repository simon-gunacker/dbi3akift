CREATE TABLE nutzer (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL,
    konto_guthaben DECIMAL(10,2) DEFAULT 0
);

CREATE TABLE medium (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    titel TEXT NOT NULL,
    typ TEXT CHECK (typ IN ('buch', 'ebook')),
    bestand INTEGER DEFAULT 0
);

CREATE TABLE ausleihe (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    nutzer_id INTEGER,
    medium_id INTEGER,
    startdatum DATE DEFAULT (DATE('now')),
    rueckgabedatum DATE DEFAULT (DATE('now', '+14 day')),
    status TEXT DEFAULT 'aktiv',  -- aktiv, beendet, gemahnt, gesperrt
    FOREIGN KEY (nutzer_id) REFERENCES nutzer(id),
    FOREIGN KEY (medium_id) REFERENCES medium(id)
);

CREATE TABLE mahnung (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    ausleihe_id INTEGER,
    betrag DECIMAL(10,2) NOT NULL CHECK (betrag > 0),
    datum DATE DEFAULT (DATE('now')),
    FOREIGN KEY (ausleihe_id) REFERENCES ausleihe(id)
);

CREATE TABLE system_log (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    aktion TEXT,
    info TEXT,
    datum DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- das Guthaben könnte sonst nie negativ werden
-- Annahme ein externen Skript checkt die Rückgabendaten und gändert den Status der Ausleihe um 
CREATE TRIGGER mahnung_verfassen
AFTER UPDATE OF status ON ausleihe
FOR EACH ROW
WHEN NEW.status = 'gemahnt'
BEGIN
    -- abziehen der Mahngebühren des Guthabens des Nutzers
    UPDATE nutzer
    SET konto_guthaben = konto_guthaben - 10
    WHERE id = NEW.nutzer_id;

    -- aufzeichnung der Mahnung
    INSERT INTO mahnung (ausleihe_id, betrag)
    VALUES (NEW.id, 10);

    -- Aufgabe 5 
    INSERT INTO system_log (aktion, info)
    VALUES ('MAHNUNG', (SELECT name 
                            FROM nutzer 
                            WHERE id = NEW.nutzer_id) || ' - Ausleihe ' || NEW.id);
END;

--Aufgabe 3

CREATE VIEW ebook_ausleihe AS
SELECT n.name  AS nutzername, m.titel AS titel,
    CAST(julianday(a.rueckgabedatum) - julianday(a.startdatum) AS INTEGER) AS dauer
FROM ausleihe a
JOIN nutzer n ON n.id = a.nutzer_id
JOIN medium m ON m.id = a.medium_id
WHERE m.typ = 'ebook';

CREATE TRIGGER ebook_ausleihe_einfuegen
INSTEAD OF INSERT ON ebook_ausleihe 
FOR EACH ROW
BEGIN
    -- nur ebooks sind erlaubt
    SELECT RAISE (ABORT, 'Kein E-Book mit diesem Titel gefunden')
    WHERE NOT EXISTS (SELECT 1 FROM medium
                    WHERE titel = NEW.titel AND typ = 'ebook');
    
    -- der Nutzer muss existieren 
    SELECT RAISE (ABORT, 'Nutzer nicht gefunden')
    WHERE NOT EXISTS (SELECT 1 FROM nutzer WHERE name = NEW.nutzername);
    
    --ausfüllen der fehlenden Daten
    INSERT INTO ausleihe (nutzer_id, medium_id, rueckgabedatum)
    VALUES (
        (SELECT id FROM nutzer WHERE name = NEW.nutzername),
        (SELECT id FROM medium WHERE titel = NEW.titel AND typ = 'ebook'),
        DATE('now', '+' || NEW.dauer || ' day')
    );

END;

-- Aufgabe 4
-- Es wird nichts gelöscht! Mann sollte Daten aus einer Datenbank nicht löschen
-- Nutzer wird nur geupdated
-- Nutzer bezogenen Daten werden umgeändert

CREATE TRIGGER loeschen_user
BEFORE DELETE ON nutzer
FOR EACH ROW
BEGIN
    -- Nutzer wir nicht gelöscht sondern nur geupdated, Name wir unkenntlich
    -- Kontoguthaben wir auf 0 gesetzt
    -- Annamhe der Kunde hat schon alles zurück gebraucht nach seinem Austretten

    UPDATE nutzer SET name = 'Gelöschter Nutzer', konto_guthaben = 0 WHERE id = OLD.id;

    -- Eintrag ins systemlog von den Daten
    INSERT INTO system_log (aktion, info)
    VALUES ('NUTZER_AUSGETRETTEN', OLD.name);

    -- Raise IGNORE muss am schluss hin sonst hört der Trigger hier gleich auf
    -- IGNORE überspringt den eigentlichen DELETE aber übernimmt die änderungen darüber
    SELECT RAISE (IGNORE);
END;

-- Aufgabe 6
-- Hier zu hab ich im shema.sql einen CHECK in der mahnung tabelle auf betrag gemacht
-- betrag DECIMAL(10,2) CHECK (betrag > 0) aber wir machen das mit einen Trigger
-- -> CHECK und NOT NULL wieder aus shema.sql entfernt, sonst würden sie vor dem Trigger greifen

CREATE TRIGGER pruefe_mahnbetrag
BEFORE INSERT ON mahnung
FOR EACH ROW
-- 0 €, negative Beträge und fehlende Beträge (NULL) werden blockiert
WHEN NEW.betrag <= 0 OR NEW.betrag IS NULL
BEGIN
    SELECT RAISE (ABORT, 'Bitte gib ein Mahnngsbetrag ein das über 0€ ist');
END;
